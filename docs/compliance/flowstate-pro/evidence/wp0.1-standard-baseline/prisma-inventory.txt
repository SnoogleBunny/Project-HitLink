Rules
- Input files: `git ls-files` entries with extensions .ts, .tsx, .js, .jsx, .mjs, or .cjs.
- Production exclusions: paths under tests/test/__tests__/fixtures/scripts; files named *.test.* or *.spec.*; and executable seed/verification files under packages/db/prisma.
- Delegate expression: a call whose source text, across whitespace/newlines, matches `(prisma|db|tx).<schema delegate>.<method>(`, where delegates are generated from current Prisma model names by lowercasing the first character.
- The parser counts direct call expressions, including multiline calls. It does not prove reachability, distinguish runtime branches, detect aliases other than prisma/db/tx, or classify calls hidden behind helper functions. These are static call-site counts, not query execution counts.
- Raw SQL search covers call and tagged-template forms of `$queryRaw`, `$queryRawUnsafe`, `$executeRaw`, and `$executeRawUnsafe` in the same tracked code extensions.

  python3 - <<'PY'
  from pathlib import Path
  import re, subprocess

  root = Path('.')
  tracked = subprocess.check_output(['git', 'ls-files'], text=True).splitlines()
  extensions = {'.ts', '.tsx', '.js', '.jsx', '.mjs', '.cjs'}
  schema = (root / 'packages/db/prisma/schema.prisma').read_text()
  models = re.findall(r'^model\s+(\w+)\s*\{', schema, re.M)
  delegates = {model[0].lower() + model[1:] for model in models}
  delegate_call = re.compile(
      r'\b(?:prisma|db|tx)\s*\.\s*('
      + '|'.join(map(re.escape, sorted(delegates, key=len, reverse=True)))
      + r')\s*\.\s*[A-Za-z_$][\w$]*\s*\(',
      re.S,
  )
  raw_sql = re.compile(
      r'\.(?:\$queryRaw|\$queryRawUnsafe|\$executeRaw|\$executeRawUnsafe)\s*(?:\(|`)'
  )

  def excluded(path):
      file = Path(path)
      return (
          bool(set(file.parts) & {'test', 'tests', '__tests__', 'fixture', 'fixtures', 'script', 'scripts'})
          or re.search(r'\.(?:test|spec)\.[^.]+$', file.name) is not None
          or path in {
              'packages/db/prisma/seed-demo.mjs',
              'packages/db/prisma/verify-demo-seed.mjs',
          }
      )

  production = {}
  excluded_calls = {}
  raw_calls = {}
  for path in tracked:
      if Path(path).suffix not in extensions:
          continue
      source = (root / path).read_text(errors='ignore')
      delegate_count = len(delegate_call.findall(source))
      raw_count = len(raw_sql.findall(source))
      if delegate_count:
          (excluded_calls if excluded(path) else production)[path] = delegate_count
      if raw_count:
          raw_calls[path] = raw_count

  outside = {path: count for path, count in production.items() if not path.startswith('packages/db/')}
  print(f'production_calls={sum(production.values())} production_files={len(production)}')
  print(f'outside_db_calls={sum(outside.values())} outside_db_files={len(outside)}')
  print(f'excluded_calls={sum(excluded_calls.values())} excluded_files={len(excluded_calls)}')
  print(f'raw_sql_calls={sum(raw_calls.values())} raw_sql_files={len(raw_calls)}')
  print('Production calls by file')
  for path, count in sorted(production.items()):
      print(f'{count:3d} {path}')
  print('Excluded calls by file')
  for path, count in sorted(excluded_calls.items()):
      print(f'{count:3d} {path}')
  print('Raw SQL calls by file')
  for path, count in sorted(raw_calls.items()):
      print(f'{count:3d} {path}')
